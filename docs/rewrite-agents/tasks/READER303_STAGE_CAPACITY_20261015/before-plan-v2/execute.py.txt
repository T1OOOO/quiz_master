def read(p):return json.loads(p.read_bytes())
def digest(p):
    assert p.is_file() and not p.is_symlink()
    before=p.stat();assert stat.S_ISREG(before.st_mode) and before.st_nlink==1
    with p.open('rb') as stream:value=hashlib.file_digest(stream,'sha256').hexdigest()
    after=p.stat();assert (before.st_dev,before.st_ino,before.st_size,before.st_mtime_ns)==(after.st_dev,after.st_ino,after.st_size,after.st_mtime_ns)
    return value

def verify_review(proof,inputs_sha,script_sha):
    assert proof.get('actual_cli_exit')==0 and proof.get('actual_hub_state')=='ACCEPTED'
    item=proof.get('actual_reviews',{}).get(TASK,{})
    args=item.get('call',{}).get('arguments',{});reply=item.get('actual_response',{})
    assert item.get('call',{}).get('action')=='review' and args.get('task_id')==TASK and args.get('verdict')=='accept' and args.get('error_count')==0
    assert reply.get('id')==TASK and reply.get('state')=='ACCEPTED' and reply.get('assignee')==AUTHOR
    assert reply.get('reviewer') and reply['reviewer']!=AUTHOR
    assert 'stage-offload-inputs sha256 '+inputs_sha in args.get('note','')
    assert 'stage-offload-script sha256 '+script_sha in args.get('note','')

def verify_manifest(inputs):
    assert set(inputs)=={str((DOCS/n).resolve()) for n in NAMES}|{str(GUARD.resolve())}
    for p,sha in inputs.items():assert re.fullmatch('[0-9a-f]{64}',sha) and digest(Path(p))==sha,'reviewed input drift: '+p

        remote=c['remote'];assert remote in expected and remote not in seen;seen.add(remote);e=expected[remote]
        local=Path(c['local']);name=PurePosixPath(tree['root']).name+'.tar.gz'
        assert local.parent==COLD and local.resolve().parent==COLD.resolve() and local.name==name==PurePosixPath(remote).name
        assert local.stat().st_size==e['size']==c['bytes'] and digest(local)==e['sha256']==c['sha256']
        assert c['actual_exit']==0 and c['members_verified']==len(tree['entries']) and c['permissions_and_ownership_verified'] is True and c['full_gzip_crc_verified'] is True
    old=receipt['original_failed_package'];assert Path(old['local'])==OLD_COPY and old['remote'] in expected and old['remote'] not in seen
