package main

import (
 "encoding/json"
 "os"
 "path/filepath"
 "quiz_master/next/server/internal/content"
)

func main() {
 for _, topic := range []string{"cinema", "music"} {
  dir := filepath.Join("next/content", topic+"-encore-20261011")
  raw, err := os.ReadFile(filepath.Join(dir,"draft.json")); if err != nil { panic(err) }
  var d content.Draft; if err=json.Unmarshal(raw,&d); err != nil { panic(err) }
  folder := "Cinema/People"; if topic=="music" { folder="MusicPeople" }
  for i := range d.Questions { d.Questions[i].Source["uri"]="/app/quizzes/"+folder+"/"+topic+"_encore_20261011.json" }
  content.Rehash(&d)
  b, err := content.Build(d,"2026.10.11.encore","2026-10-11T00:00:00Z"); if err != nil { panic(err) }
  if err=content.WriteJSON(filepath.Join(dir,"draft.json"),d,true); err != nil { panic(err) }
  if err=content.WriteJSON(filepath.Join(dir,"bundle.json"),b,true); err != nil { panic(err) }
 }
}
